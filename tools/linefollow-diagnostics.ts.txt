// ── Line follower for tigez ───────────────────────────────────
// Sensor: ElecFreaks PlanetX Trackbit — four reflectance channels on
// I2C 0x1A, mounted ahead of the wheels with CHANNEL 0 ON THE LEFT.
// Register 4 returns the line bitmask: bit i set when channel i sees
// the line. One write + one read gets all four channels, which is why
// this reads the register directly rather than calling the PlanetX
// per-channel block four times.
//
// Steering is proportional: yawRate = KP * error, with yaw CCW-positive
// and the weights positive on the left — so a line drifting left yields
// a positive error and turns the robot left, back onto it.
//
// driveTwist() is a REPLACING command: it sets a wheel-velocity demand
// under a lease and returns, so each pass of the loop supersedes the
// previous demand rather than queueing behind it. driveTick() advances
// one control cycle and self-paces at ~24 ms, so the loop needs no
// pause() of its own. If the loop ever stops ticking, the drive's own
// watchdog neutralises the wheels within ~150 ms — a crash or a break
// cannot leave the robot driving.
const TRACKBIT = 0x1a
const WEIGHT = [1.5, 0.5, -0.5, -1.5]   // channel 0 = left
const NOTHING_SEEN = 999

let running = false
let aborted = false

function lineBits(): number {
    pins.i2cWriteNumber(TRACKBIT, 4, NumberFormat.Int8LE)
    return pins.i2cReadNumber(TRACKBIT, NumberFormat.UInt8LE, false) & 0x0f
}

// Where the line sits under the array, in channel units:
// +1.5 = far left only, -1.5 = far right only, 0 = centred.
function lineError(bits: number): number {
    let sum = 0
    let n = 0
    for (let i = 0; i < 4; i++) {
        if (bits & (1 << i)) { sum += WEIGHT[i]; n += 1 }
    }
    return n == 0 ? NOTHING_SEEN : sum / n
}

function followLine(speed: number, maxS: number, kp: number) {
    if (running) { diffDrive.emitLine("LINE:already-running"); return }
    running = true
    aborted = false
    let lastErr = 0
    let lostSince = -1
    let reason = "time"
    let t0 = control.millis()
    diffDrive.emitLine("LINE:start speed=" + speed + " maxS=" + maxS + " kp=" + kp)
    basic.showIcon(IconNames.Diamond)

    diffDrive.driveTwist(speed, 0)
    while (diffDrive.driveTick()) {
        let now = control.millis() - t0
        if (aborted) { reason = "abort"; break }
        if (now > maxS * 1000) break
        let err = lineError(lineBits())
        if (err == NOTHING_SEEN) {
            // Nothing seen: sweep back toward the side the line was last
            // on, at reduced speed, and give up after 1.5 s.
            if (lostSince < 0) lostSince = now
            else if (now - lostSince > 1500) { reason = "lost"; break }
            diffDrive.driveTwist(speed * 0.4, lastErr >= 0 ? 90 : -90)
        } else {
            lostSince = -1
            if (err != 0) lastErr = err
            diffDrive.driveTwist(speed, kp * err)
        }
    }
    diffDrive.stopMove()
    basic.clearScreen()
    diffDrive.emitLine("LINE:end:" + reason + ":" + (control.millis() - t0)
        + " x=" + Math.round(diffDrive.poseX())
        + " y=" + Math.round(diffDrive.poseY()))
    running = false
}

// ── Buttons ───────────────────────────────────────────────────
input.onButtonPressed(Button.A, function () {
    diffDrive.emitLine("btn A")
    followLine(8, 90, 60)
})
input.onButtonPressed(Button.AB, function () {
    diffDrive.emitLine("btn AB")
    aborted = true
})

// ── Remote triggers ───────────────────────────────────────────
// RUN:line[:speed[:maxS[:kp]]]   defaults 8 cm/s, 90 s, kp 60
diffDrive.onRun("line", function (arg) {
    let speed = diffDrive.runArgCount() > 0 ? diffDrive.runArg(0) : 8
    let maxS = diffDrive.runArgCount() > 1 ? diffDrive.runArg(1) : 90
    let kp = diffDrive.runArgCount() > 2 ? diffDrive.runArg(2) : 60
    followLine(speed, maxS, kp)
})
diffDrive.onRun("abort", function (arg) { aborted = true })
// RUN:a raises the real button-A event, so the bench exercises the same
// path a finger does.
diffDrive.onRun("a", function (arg) {
    diffDrive.emitLine("raise A")
    control.raiseEvent(EventBusSource.MICROBIT_ID_BUTTON_A,
        EventBusValue.MICROBIT_BUTTON_EVT_CLICK)
})
// Orientation / calibration check: which channels see the line, right now.
diffDrive.onRun("linesense", function (arg) {
    let n = arg > 0 ? arg : 20
    for (let i = 0; i < n; i++) {
        let b = lineBits()
        diffDrive.emitLine("TB:" + b
            + " ch=" + (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1) + ((b >> 3) & 1)
            + " err=" + lineError(b))
        basic.pause(150)
    }
})
// ── Calibration: make the robot tell us its own geometry ──────
// Rotating IN PLACE cannot translate the robot off the mat, so this is
// the safe way to learn three unknowns at once: whether the sensor
// works, which channel order it reports, and which way the robot faces
// relative to the lines around it. Streams the bitmask the whole time.
diffDrive.onRun("sweep", function (arg) {
    let degPerS = arg > 0 ? arg : 40
    let secs = diffDrive.runArgCount() > 1 ? diffDrive.runArg(1) : 9
    diffDrive.emitLine("SWEEP:start rate=" + degPerS + " secs=" + secs)
    let t0 = control.millis()
    let last = -1
    diffDrive.driveTwist(0, degPerS)
    while (diffDrive.driveTick()) {
        let now = control.millis() - t0
        if (now > secs * 1000) break
        if (aborted) break
        let b = lineBits()
        if (b != last) {
            diffDrive.emitLine("SWEEP:" + now + "ms bits=" + b
                + " ch=" + (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1) + ((b >> 3) & 1)
                + " heading=" + Math.round(diffDrive.heading()))
            last = b
        }
        diffDrive.driveTwist(0, degPerS)
    }
    diffDrive.stopMove()
    diffDrive.emitLine("SWEEP:end heading=" + Math.round(diffDrive.heading()))
})

// Creep forward a little, streaming the sensor. Small and time-bounded.
diffDrive.onRun("nudge", function (arg) {
    let cm = arg > 0 ? arg : 6
    diffDrive.emitLine("NUDGE:start " + cm + "cm")
    diffDrive.resetPose()
    let t0 = control.millis()
    let last = -1
    diffDrive.driveTwist(5, 0)
    while (diffDrive.driveTick()) {
        if (aborted) break
        if (control.millis() - t0 > 12000) break
        if (diffDrive.poseX() >= cm) break
        let b = lineBits()
        if (b != last) {
            diffDrive.emitLine("NUDGE:x=" + Math.round(diffDrive.poseX())
                + " bits=" + b
                + " ch=" + (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1) + ((b >> 3) & 1))
            last = b
        }
        diffDrive.driveTwist(5, 0)
    }
    diffDrive.stopMove()
    diffDrive.emitLine("NUDGE:end x=" + Math.round(diffDrive.poseX()))
})

// ── Wheel check ───────────────────────────────────────────────
// A driveTwist(0, w) is meant to be a pure spin: wheelsV(-t, +t). If the
// robot TRANSLATES instead, the two sides are not matched and every
// steering command is really an arc — which no amount of line-following
// gain can fix. Drive each wheel alone, briefly, and compare the pose
// each one produces. Deliberately short and slow: the robot is near the
// edge of the mat.
function runWheel(left: number, right: number, ms: number, tag: string) {
    diffDrive.resetPose()
    let t0 = control.millis()
    diffDrive.setWheelSpeeds(left, right)
    while (diffDrive.driveTick()) {
        if (aborted) break
        if (control.millis() - t0 > ms) break
        diffDrive.setWheelSpeeds(left, right)
    }
    diffDrive.stopMove()
    basic.pause(400)
    diffDrive.emitLine("WHEEL:" + tag
        + " x=" + Math.round(diffDrive.poseX())
        + " y=" + Math.round(diffDrive.poseY())
        + " heading=" + Math.round(diffDrive.heading()))
}

diffDrive.onRun("wheeltest", function (arg) {
    let v = arg > 0 ? arg : 5
    diffDrive.emitLine("WHEELTEST:start v=" + v)
    runWheel(v, 0, 900, "left-only")
    basic.pause(600)
    runWheel(0, v, 900, "right-only")
    basic.pause(600)
    runWheel(v, v, 900, "both")
    diffDrive.emitLine("WHEELTEST:end")
})

diffDrive.onRunCommand(function (name, arg) {
    diffDrive.emitLine("run rx name=" + name + " arg=" + arg)
})

diffDrive.emitLine("boot line-follower ready")
basic.showIcon(IconNames.Ghost)
